-- V2-H — 0003_lang_tags.sql · DATA migration for EXISTING DBs (seedIfEmpty
-- won't rerun). Restructures Languages into per-language groups, retags the
-- 3 existing tasks with lang/phase tags, inserts the 9 new tasks.
-- Run once in the Supabase SQL editor. Idempotence guard: aborts if a
-- 'Japanese' group already exists under Languages.
DO $$
DECLARE
  lang_id uuid;
  ja_id uuid; zh_id uuid; my_id uuid;
BEGIN
  SELECT id INTO lang_id FROM roadmap_tasks WHERE kind = 'group' AND title = 'Languages' LIMIT 1;
  IF lang_id IS NULL THEN RAISE EXCEPTION 'Languages group not found'; END IF;
  IF EXISTS (SELECT 1 FROM roadmap_tasks WHERE parent_id = lang_id AND kind = 'group' AND title = 'Japanese') THEN
    RAISE EXCEPTION 'Already migrated (Japanese group exists)';
  END IF;

  INSERT INTO roadmap_tasks (parent_id, title, kind, position, expanded, tags)
    VALUES (lang_id, 'Japanese', 'group', 0, true, '{}') RETURNING id INTO ja_id;
  INSERT INTO roadmap_tasks (parent_id, title, kind, position, expanded, tags)
    VALUES (lang_id, 'Chinese', 'group', 1, true, '{}') RETURNING id INTO zh_id;
  INSERT INTO roadmap_tasks (parent_id, title, kind, position, expanded, tags)
    VALUES (lang_id, 'Burmese', 'group', 2, true, '{}') RETURNING id INTO my_id;

  -- reparent + retag the three existing tasks
  UPDATE roadmap_tasks SET parent_id = ja_id, position = 3,
    tags = '{lang,ja,ph-listening,train,audio-only}'
    WHERE kind = 'task' AND title = 'Japanese N1 listening practice';
  UPDATE roadmap_tasks SET parent_id = zh_id, position = 1,
    tags = '{lang,zh,ph-quick,train,audio-only,short-burst,daily-routine}'
    WHERE kind = 'task' AND title = 'Chinese HSK-1, 5 words today';
  UPDATE roadmap_tasks SET parent_id = zh_id, position = 4,
    tags = '{lang,zh,ph-speaking,home,keyboard}'
    WHERE kind = 'task' AND title = 'Chinese conversation Excel update';

  -- new tasks (positions interleave with the moves above)
  INSERT INTO roadmap_tasks (parent_id, title, kind, done, position, tags) VALUES
    (ja_id, 'Japanese app study · 5 min',        'task', false, 0, '{lang,ja,ph-quick,train,short-burst,daily-routine}'),
    (ja_id, 'NHK Easy article · 1',              'task', false, 1, '{lang,ja,ph-reading,train,short-burst}'),
    (ja_id, 'Writing test · 5 min',              'task', false, 2, '{lang,ja,ph-writing,home,keyboard,short-burst}'),
    (ja_id, 'Shadowing · 10 min',                'task', false, 4, '{lang,ja,ph-speaking,home,audio-only}'),
    (zh_id, 'Chinese app study · 5 min',         'task', false, 0, '{lang,zh,ph-quick,train,short-burst,daily-routine}'),
    (zh_id, 'Graded reader · ch. 2',             'task', false, 2, '{lang,zh,ph-reading,train,short-burst}'),
    (zh_id, 'HSK-1 audio drill',                 'task', false, 3, '{lang,zh,ph-listening,train,audio-only}'),
    (my_id, 'Burmese app study · 5 min',         'task', false, 0, '{lang,my,ph-quick,train,short-burst,daily-routine}'),
    (my_id, 'Script practice · 1 row',           'task', false, 1, '{lang,my,ph-writing,home,short-burst}');
END $$;
