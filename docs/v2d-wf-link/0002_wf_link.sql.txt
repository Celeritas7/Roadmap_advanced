-- 2 · Link table: one row per mirrored pair
create table if not exists task_links (
  id uuid primary key default gen_random_uuid(),
  wf_task_id uuid not null unique
    references weekly_focus_tasks (id) on delete cascade,   -- ⚠PROBE table name
  rm_task_id uuid not null unique
    references roadmap_tasks (id) on delete cascade,        -- ⚠PROBE table name
  source text not null check (source in ('wf', 'rm')),
  last_synced_at timestamptz not null default now()
);
