// V2-D · Phase 2 — src/lib/wfLink.ts (Roadmap side)
// DRAFT: ⚠PROBE marks assumptions to confirm in Phase 0 (WF table/column
// names, week-scoping field, category values).
// Pull-merge sync: WF open-this-week tasks → mirrored Roadmap tasks via
// task_links; shared fields (title, done, '@…' contexts) merged by
// last-write-wins on updated_at. Roadmap-private structure (parent/position/
// project tag) never leaves this app; WF-private fields never enter it.

import { supabase } from './supabase.ts'
import type { TaskRow } from '../types.ts'

// ⚠PROBE: real WF row shape
type WfTask = {
  id: string
  title: string
  done: boolean
  category: string | null   // ⚠PROBE: field name (category? topic? board?)
  contexts: string[]
  updated_at: string
}

type LinkRow = {
  id: string
  wf_task_id: string
  rm_task_id: string
  source: 'wf' | 'rm'
  last_synced_at: string
}

// Topic → Roadmap project id. ⚠PROBE WF's real category values; 'weekly' is
// the catch-all project (add to seed if missing). Vetoable design default.
const TOPIC_TO_PROJECT: Record<string, string> = {
  PRODUCTIVITY: 'dx',
  CODING: 'dx',
  GENERAL_PURPOSE: 'weekly',
  ACADEMICS: 'lang',
  AI_AND_MATHS: 'lang',
  FINANCE: 'weekly',
}
const projectFor = (c: string | null) => TOPIC_TO_PROJECT[c ?? ''] ?? 'weekly'

const isContext = (tag: string) => tag.startsWith('@')

export type WfSyncResult = { imported: number; merged: number }

export async function syncWeekly(rmTasks: TaskRow[]): Promise<WfSyncResult> {
  // ⚠PROBE: the "open this week" filter — WF's week/board scoping column.
  const [{ data: wfTasks, error: e1 }, { data: links, error: e2 }] = await Promise.all([
    supabase.from('weekly_focus_tasks').select('*'), // ⚠PROBE + .eq(weekField, currentWeek)
    supabase.from('task_links').select('*'),
  ])
  if (e1) throw e1
  if (e2) throw e2

  const linkByWf = new Map((links as LinkRow[]).map((l) => [l.wf_task_id, l]))
  const rmById = new Map(rmTasks.map((t) => [t.id, t]))
  let imported = 0
  let merged = 0

  for (const wf of (wfTasks ?? []) as WfTask[]) {
    const link = linkByWf.get(wf.id)

    if (!link) {
      // New WF task → mirror into Roadmap under the topic-mapped folder.
      const proj = projectFor(wf.category)
      const { data: rmRow, error } = await supabase
        .from('roadmap_tasks') // ⚠PROBE table name
        .insert({
          parent_id: null, // Phase-0 option: folder group id per project
          title: wf.title,
          kind: 'task',
          position: 9999,
          done: wf.done,
          tags: [proj, ...wf.contexts.filter(isContext)],
        })
        .select()
        .single()
      if (error) throw error
      const { error: le } = await supabase.from('task_links').insert({
        wf_task_id: wf.id,
        rm_task_id: (rmRow as TaskRow).id,
        source: 'wf',
      })
      if (le) throw le
      imported++
      continue
    }

    // Linked pair → last write wins on shared fields.
    const rm = rmById.get(link.rm_task_id)
    if (!rm) continue // RM side deleted → orphan; leave for cleanup pass
    const wfT = Date.parse(wf.updated_at)
    const rmT = Date.parse(rm.updated_at)
    const syncedT = Date.parse(link.last_synced_at)
    if (wfT <= syncedT && rmT <= syncedT) continue // nothing new

    if (wfT >= rmT) {
      // WF newer → copy shared fields into RM (keep RM-private tags: project id)
      const keep = rm.tags.filter((t) => !isContext(t))
      const { error } = await supabase
        .from('roadmap_tasks') // ⚠PROBE
        .update({ title: wf.title, done: wf.done, tags: [...keep, ...wf.contexts.filter(isContext)] })
        .eq('id', rm.id)
      if (error) throw error
    } else {
      // RM newer → copy shared fields into WF
      const { error } = await supabase
        .from('weekly_focus_tasks') // ⚠PROBE
        .update({ title: rm.title, done: rm.done, contexts: rm.tags.filter(isContext) })
        .eq('id', wf.id)
      if (error) throw error
    }
    const { error: te } = await supabase
      .from('task_links')
      .update({ last_synced_at: new Date().toISOString() })
      .eq('id', link.id)
    if (te) throw te
    merged++
  }
  return { imported, merged }
}
